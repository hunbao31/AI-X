'use client';
import { InlineMath, BlockMath } from 'react-katex';

// Component này nhận vào 1 đoạn text CÓ THỂ chứa LaTeX
// (đánh dấu bằng $...$  cho công thức trên 1 dòng,
//  hoặc $$...$$  cho công thức chiếm riêng 1 dòng)
// và tự động tách ra để hiển thị đúng, phần chữ thường
// vẫn hiển thị bình thường.

export default function MathText({ text }: { text: string }) {
  if (!text) return null;

  // Tách chuỗi theo dấu $ hoặc $$
  const parts = text.split(/(\$\$[^$]+\$\$|\$[^$]+\$)/g);

  return (
    <span>
      {parts.map((part, i) => {
        if (part.startsWith('$$') && part.endsWith('$$')) {
          return <BlockMath key={i} math={part.slice(2, -2)} />;
        }
        if (part.startsWith('$') && part.endsWith('$')) {
          return <InlineMath key={i} math={part.slice(1, -1)} />;
        }
        return <span key={i}>{part}</span>;
      })}
    </span>
  );
}
